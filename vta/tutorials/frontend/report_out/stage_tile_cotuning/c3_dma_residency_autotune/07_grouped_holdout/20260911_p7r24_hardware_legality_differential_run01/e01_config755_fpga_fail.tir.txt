@I.ir_module
class Module:
    @T.prim_func
    def main(placeholder: T.Buffer((1, 12, 21, 21, 1, 16), "int8"), placeholder_1: T.Buffer((20, 12, 1, 1, 16, 16), "int8"), compute: T.Buffer((1, 20, 21, 21, 1, 16), "int8")):
        T.func_attr({"from_legacy_te_schedule": T.bool(True), "global_symbol": "main", "tir.noalias": T.bool(True)})
        T.tir.vta.coproc_dep_push(3, 2)
        T.tir.vta.coproc_dep_push(3, 2)
        for i1_outer_outer, i3_outer in T.grid(5, 21):
            for cthread_s in range(2):
                vta = T.int32()
                T.attr(T.iter_var(vta, None, "ThreadIndex", "vta"), "coproc_scope", 2)
                T.tir.vta.coproc_dep_pop(3, 2)
                with T.attr(T.iter_var(vta, None, "ThreadIndex", "vta"), "coproc_uop_scope", "VTAPushGEMMOp"):
                    T.call_extern("int32", "VTAUopLoopBegin", 2, 21, 0, 0)
                    T.call_extern("int32", "VTAUopLoopBegin", 21, 1, 0, 0)
                    T.tir.vta.uop_push(0, 1, cthread_s * 42, 0, 0, 0, 0, 0)
                    T.call_extern("int32", "VTAUopLoopEnd")
                    T.call_extern("int32", "VTAUopLoopEnd")
                T.tir.vta.coproc_dep_push(2, 1)
            vta = T.int32()
            with T.attr(T.iter_var(vta, None, "ThreadIndex", "vta"), "coproc_scope", 1):
                T.tir.vta.coproc_dep_pop(2, 1)
                T.call_extern("int32", "VTALoadBuffer2D", T.tvm_thread_context(T.tir.vta.command_handle()), placeholder.data, i3_outer, 1, 252, 21, 0, 0, 0, 0, 0, 2)
                T.call_extern("int32", "VTALoadBuffer2D", T.tvm_thread_context(T.tir.vta.command_handle()), placeholder_1.data, i1_outer_outer * 48, 24, 1, 24, 0, 0, 0, 0, 0, 1)
                T.tir.vta.coproc_dep_push(1, 2)
            with T.attr(T.iter_var(vta, None, "ThreadIndex", "vta"), "coproc_scope", 1):
                T.tir.vta.coproc_dep_pop(2, 1)
                T.call_extern("int32", "VTALoadBuffer2D", T.tvm_thread_context(T.tir.vta.command_handle()), placeholder.data, i3_outer, 1, 252, 21, 0, 0, 0, 0, 252, 2)
                T.call_extern("int32", "VTALoadBuffer2D", T.tvm_thread_context(T.tir.vta.command_handle()), placeholder_1.data, i1_outer_outer * 48 + 24, 24, 1, 24, 0, 0, 0, 0, 24, 1)
                T.tir.vta.coproc_dep_push(1, 2)
            for cthread_s in range(2):
                T.attr(T.iter_var(vta, None, "ThreadIndex", "vta"), "coproc_scope", 2)
                T.tir.vta.coproc_dep_pop(1, 2)
                with T.attr(T.iter_var(vta, None, "ThreadIndex", "vta"), "coproc_uop_scope", "VTAPushGEMMOp"):
                    T.call_extern("int32", "VTAUopLoopBegin", 12, 0, 21, 1)
                    T.call_extern("int32", "VTAUopLoopBegin", 2, 21, 0, 12)
                    for i in range(21):
                        T.tir.vta.uop_push(0, 0, cthread_s * 42 + i, cthread_s * 252 + i, cthread_s * 24, 0, 0, 0)
                    T.call_extern("int32", "VTAUopLoopEnd")
                    T.call_extern("int32", "VTAUopLoopEnd")
                with T.attr(T.iter_var(vta, None, "ThreadIndex", "vta"), "coproc_uop_scope", "VTAPushALUOp"):
                    T.call_extern("int32", "VTAUopLoopBegin", 42, 1, 1, 0)
                    T.tir.vta.uop_push(1, 0, cthread_s * 42, cthread_s * 42, 0, 3, 1, 8)
                    T.call_extern("int32", "VTAUopLoopEnd")
                with T.attr(T.iter_var(vta, None, "ThreadIndex", "vta"), "coproc_uop_scope", "VTAPushALUOp"):
                    T.call_extern("int32", "VTAUopLoopBegin", 42, 1, 1, 0)
                    T.tir.vta.uop_push(1, 0, cthread_s * 42, cthread_s * 42, 0, 0, 1, 127)
                    T.call_extern("int32", "VTAUopLoopEnd")
                with T.attr(T.iter_var(vta, None, "ThreadIndex", "vta"), "coproc_uop_scope", "VTAPushALUOp"):
                    T.call_extern("int32", "VTAUopLoopBegin", 42, 1, 1, 0)
                    T.tir.vta.uop_push(1, 0, cthread_s * 42, cthread_s * 42, 0, 1, 1, 0)
                    T.call_extern("int32", "VTAUopLoopEnd")
                T.tir.vta.coproc_dep_push(2, 3)
            with T.attr(T.iter_var(vta, None, "ThreadIndex", "vta"), "coproc_scope", 3):
                T.tir.vta.coproc_dep_pop(2, 3)
                T.call_extern("int32", "VTAStoreBuffer2D", T.tvm_thread_context(T.tir.vta.command_handle()), 0, 4, compute.data, i1_outer_outer * 1764 + i3_outer, 1, 42, 21)
                T.tir.vta.coproc_dep_push(3, 2)
            T.attr(T.iter_var(vta, None, "ThreadIndex", "vta"), "coproc_scope", 3)
            T.tir.vta.coproc_dep_pop(2, 3)
            T.call_extern("int32", "VTAStoreBuffer2D", T.tvm_thread_context(T.tir.vta.command_handle()), 42, 4, compute.data, i1_outer_outer * 1764 + i3_outer + 882, 1, 42, 21)
            T.tir.vta.coproc_dep_push(3, 2)
        T.tir.vta.coproc_dep_pop(3, 2)
        T.tir.vta.coproc_dep_pop(3, 2)
        T.tir.vta.coproc_sync()
