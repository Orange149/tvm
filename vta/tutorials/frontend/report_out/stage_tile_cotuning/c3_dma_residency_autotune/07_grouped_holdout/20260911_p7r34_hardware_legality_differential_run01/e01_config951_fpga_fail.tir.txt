        T.tir.vta.coproc_dep_push(3, 2)

                    T.call_extern("int32", "VTAUopLoopEnd")

                T.tir.vta.coproc_dep_pop(2, 1)
                T.call_extern("int32", "VTALoadBuffer2D", T.tvm_thread_context(T.tir.vta.command_handle()), placeholder.data, i3_outer * 3, 3, 252, 21, 0, 0, 0, 0, 756, 2)
                T.call_extern("int32", "VTALoadBuffer2D", T.tvm_thread_context(T.tir.vta.command_handle()), placeholder_1.data, i1_outer_outer * 120 + 60, 60, 1, 60, 0, 0, 0, 0, 60, 1)

            for cthread_s in range(2):
                T.attr(T.iter_var(vta, None, "ThreadIndex", "vta"), "coproc_scope", 2)
                T.tir.vta.coproc_dep_pop(1, 2)
